// ===== (Gift Code + Thông số game) DÁN THÊM vào Firebase Console → Firestore → Rules (bên trong `match /databases/{database}/documents { ... }`) =====
// Giữ nguyên các quy tắc cũ (saves, pvp_players, chat_messages...). Chỉ THÊM phần dưới.

function isAdmin() {
  return request.auth != null
    && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
}

// Danh sách admin: tạo tay trong Console (Document ID = UID tài khoản admin). Client không được ghi.
match /admins/{uid} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow write: if false;
}

match /gift_codes/{code} {
  allow get: if request.auth != null;          // người chơi chỉ GET được từng mã, không liệt kê được
  allow list: if isAdmin();
  allow create, delete: if isAdmin();

  allow update: if isAdmin()
    // người chơi chỉ được tăng 'used' lên đúng 1, khi mã còn hiệu lực + còn lượt, và đồng thời ghi 'redeemed/{uid}' của chính mình
    || (request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['used'])
        && request.resource.data.used == resource.data.used + 1
        && resource.data.enabled == true
        && (resource.data.max == 0 || resource.data.used < resource.data.max)
        && (resource.data.exp == 0 || request.time.toMillis() < resource.data.exp)
        && !exists(/databases/$(database)/documents/gift_codes/$(code)/redeemed/$(request.auth.uid))
        && existsAfter(/databases/$(database)/documents/gift_codes/$(code)/redeemed/$(request.auth.uid)));

  match /redeemed/{uid} {
    allow read: if isAdmin() || (request.auth != null && request.auth.uid == uid);
    allow create: if request.auth != null && request.auth.uid == uid;
    allow delete: if isAdmin();
  }
}

// Thông số game do admin chỉnh (trang admin-config.html). Ai cũng ĐỌC được (game tải về khi mở); chỉ admin được GHI.
match /game_config/{doc} {
  allow read: if true;
  allow write: if isAdmin();
}


// ===== (Quản lý người chơi: khóa acc / reset / thu hồi vật phẩm / khôi phục khung giờ) — trang admin-players.html =====
// QUAN TRỌNG: THAY khối `match /saves/{uid}` cũ bằng khối dưới (giữ nguyên các khối khác), rồi THÊM 2 khối account_ctl và save_history.

function ctlOf(uid) { return /databases/$(database)/documents/account_ctl/$(uid); }
function notLocked(uid) { return !(exists(ctlOf(uid)) && get(ctlOf(uid)).data.get('locked', false) == true); }
function epochOk(uid) { return request.resource.data.get('ep', 0) == (exists(ctlOf(uid)) ? get(ctlOf(uid)).data.get('epoch', 0) : 0); }

match /saves/{uid} {
  allow get: if request.auth != null && (request.auth.uid == uid || isAdmin());
  allow list: if isAdmin();
  allow create, update: if isAdmin()
    || (request.auth != null && request.auth.uid == uid && notLocked(uid) && epochOk(uid));   // bị khóa / máy cũ sau reset -> không ghi đè được
  allow delete: if isAdmin();
}

// Khóa + epoch: chỉ admin ghi; người chơi chỉ đọc của chính mình
match /account_ctl/{uid} {
  allow get: if request.auth != null && (request.auth.uid == uid || isAdmin());
  allow list: if isAdmin();
  allow write: if isAdmin();
}

// Ảnh chụp theo giờ (client tự ghi mỗi giờ, giữ 72 giờ) để admin khôi phục
match /save_history/{uid}/snaps/{id} {
  allow read: if isAdmin();
  allow create, update: if isAdmin()
    || (request.auth != null && request.auth.uid == uid && notLocked(uid) && epochOk(uid));
  allow delete: if isAdmin() || (request.auth != null && request.auth.uid == uid);
}

// ===== (MỚI) Hộp thư cá nhân + Tỷ Võ cho admin — THÊM vào cùng chỗ (bên trong `match /databases/{database}/documents { ... }`) =====

// Thư admin gửi cho từng người chơi (trang admin.html → Người chơi → "Gửi quà / thư").
// Người chơi chỉ ĐỌC thư của mình và chỉ được đổi claimed từ false -> true khi nhận quà (không sửa được nội dung quà).
match /mail_inbox/{uid}/msgs/{id} {
  allow read: if request.auth != null && (request.auth.uid == uid || isAdmin());
  allow create, delete: if isAdmin();
  allow update: if isAdmin()
    || (request.auth != null && request.auth.uid == uid
        && resource.data.get('claimed', false) == false
        && request.resource.data.claimed == true
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['claimed', 'claimedAt']));
}

// Admin được ghi vào pvp_players để đánh dấu "stale" (vừa reset / khôi phục) hoặc "locked" (khóa acc) —
// người khác sẽ không đấu Tỷ Võ với số liệu cũ của tài khoản đó. GIỮ NGUYÊN quy tắc pvp_players cũ của người chơi
// (Firestore cho phép nhiều khối match cùng đường dẫn, chỉ cần 1 khối cho phép là được).
match /pvp_players/{uid} {
  allow write: if isAdmin();
}
